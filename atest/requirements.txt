# External Python modules required by acceptance tests.
# See atest/README.rst for more information.

docutils >= 0.10
pygments

# Used by TestCheckerLibrary to validate output.xml against the schema.
# Without it the shared 'Process Output' keyword is unavailable and nearly
# every acceptance test fails in its parent suite setup.
xmlschema

pyyaml

# Python 3.13 removed the telnetlib module (PEP 594) that the standard
# Telnet library needs. This backport restores it; without it e.g. the
# whole Libdoc python_library suite fails in its setup because it
# documents robot.libraries.Telnet. If PyPI is not reachable, copying
# Lib/telnetlib.py from a Python <= 3.12 installation into site-packages
# works as well.
standard-telnetlib; python_version >= '3.13'

# On Linux installing lxml with pip may require compilation and development
# headers. Alternatively it can be installed using a package manager like
# `sudo apt-get install python-lxml`.
lxml; platform_python_implementation == 'CPython'

pillow >= 7.1.0; platform_system == 'Windows'

-r ../utest/requirements.txt
